\begin{tcolorbox}[
    colback=gray!10,
    colframe=black,
    arc=0mm
]
\small
Insertion Sort is a simple sorting algorithm that builds the final sorted array one element at a time. 
At each iteration, the current element is inserted into its correct position inside the already sorted part of the array.
\end{tcolorbox}

\begin{tcolorbox}[
    colback=black!5,
    colframe=black,
    arc=0mm,
    title=Pseudo-code
]
\small
\begin{lstlisting}
Insertion-Sort(A, n)

    for j = 2 to n
        key = A[j]
        i = j - 1

        while i > 0 and A[i] > key
            A[i + 1] = A[i]
            i = i - 1

        A[i + 1] = key
\end{lstlisting}
\end{tcolorbox}

    colback=cyan!20,
