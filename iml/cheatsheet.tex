\begin{thm}[title=Core Empirical Risk]
\[
R(w)=\frac1N\sum_{i=1}^{N}\ell(\hat y_i,y_i),\qquad
\min_w R(w)
\]
\textbf{Gradient descent:} \(w_{t+1}=w_t-\eta\nabla_w R(w_t)\).
\end{thm}
