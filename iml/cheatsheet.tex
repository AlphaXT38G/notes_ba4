\section{Notations \& Core Concepts}

\begin{prop}[title=Empirical Risk]
\[ R(w) = \frac{1}{N}\sum_{i=1}^{N} \ell(\hat{y}_i,\, y_i) \quad \Rightarrow \quad \min_w R(w) \]
\end{prop}

\begin{prop}[title=Gradient Descent]
\[ w_k \leftarrow w_{k-1} - \eta\,\nabla_w R(w_{k-1}) \]
Stop when $|R(w_k)-R(w_{k-1})|<\delta$ or $\|w_k-w_{k-1}\|<\delta$.
\end{prop}

\begin{defn}[title=Useful Identities]
$\nabla_b(a^Tb)=a$ \quad $\nabla_a\|a\|^2=2a$ \quad $\|a\|^2=a^Ta$
\end{defn}

\section{Preprocessing}

\subsection{Normalization}
\textbf{Always} apply before distance-based methods (KNN, K-Means); improves GD convergence.
\begin{defn}[title=Methods]
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lll@{}}
\textbf{Method} & \textbf{Formula} & \textbf{Result}\\
Z-score  & $\frac{x^{(d)}-\mu_d}{\sigma_d}$ & $\mu{=}0,\sigma{=}1$\\
Min-max  & $\frac{x^{(d)}-\min_d}{\max_d-\min_d}$ & $[0,1]$\\
Max norm & $\frac{x^{(d)}}{\max_d}$ & $[-1,1]$\\
\end{tabular}
\end{defn}
\textbf{Bias term:} append $1$ to $x_i$ so $D\leftarrow D+1$.

\subsection{Missing Data}
\kv{Drop}{loses info}
\kv{Global constant / mean}{fast, ignores class}
\kv{Class-conditioned mean}{best}

\subsection{Noise / Binning}
Sort $\to$ bins (equal width \textit{or} equal freq.) $\to$ replace by bin mean/median.

\subsection{Imbalanced Data}
Skewed classes $\Rightarrow$ accuracy misleading (predicting majority always gives high acc.).
\begin{defn}[title=Fixes]
\kv{Oversampling}{replicate/generate (SMOTE) minority samples}
\kv{Undersampling}{remove redundant majority samples}
\kv{Reweighting}{$R = \frac{1}{N}\sum_i \beta_i\,\ell_i,\quad \beta_i = 1 - \frac{N_k}{N}$}
\end{defn}

\subsection{Text: Bag of Words (BoW)}
Vocabulary of $D$ words $\to$ count occurrences per doc $\to$ normalize $\Rightarrow x_i\in\R^D$.\\
\textit{Limits:} no word order; stopwords dominate; sparse.

\subsection{Images: Visual BoW}
Extract patches $\to$ cluster into $D$ visual words $\to$ count $\to$ normalize $\Rightarrow x_i\in\R^D$.

\subsection{Data Augmentation}
Add samples via label-preserving transforms $\Rightarrow$ less overfitting.\\
\textit{Images:} flip, rotate, crop, affine transform, color jitter.

\cheattitle{Feature Engineering \& Dimensionality Reduction}{}{}

\section{Feature Expansion \& Regularization [R/CLA]}
\textit{When:} data has non-linear structure but you want a linear model's tractability.

\begin{defn}[title=Polynomial Expansion]
$\phi: \R^D \to \R^F$ maps $x$ to higher-dim features (e.g.\ $[1,x_1,x_2,x_1^2,x_1x_2,x_2^2]^T$).
\[ \hat{y}_i = w^T\phi(x_i), \qquad w^\star = (\Phi^T\Phi)^{-1}\Phi^Ty \]
The model is linear in $\phi(x)$ but non-linear in $x$.
\end{defn}

\begin{thm}[title=Cover's Theorem]
A complex problem is more likely linearly separable in high-dim space (provided not densely populated).
\end{thm}

\begin{warn}[title=Overfitting Risk]
High-degree $\phi \Rightarrow F\gg N \Rightarrow$ overfitting. Use regularization + cross-validation to choose degree.
\end{warn}

\subsection{L2 Regularization (Ridge)}
\begin{defn}[title=Regularized Objective]
\[ E(w) = \underbrace{R(w)}_{\text{fit}} + \underbrace{\lambda w^Tw}_{\text{penalty}} \]
$\lambda>0$ from cross-validation. Penalizes large weights $\Rightarrow$ smoother model.
\end{defn}

\begin{thm}[title=Ridge Solution (always invertible!)]
\[ w^\star = (\Phi^T\Phi + \lambda I_F)^{-1}\Phi^Ty \]
\textbf{Multi-output:} $W^\star = (\Phi^T\Phi + \lambda I_F)^{-1}\Phi^TY$
\end{thm}

\section{Dimensionality Reduction [R/CLA/CLU]}
\textit{Goal:} $x_i \in \R^D \to y_i \in \R^d$ with $d \ll D$, preserving structure.

\subsection{PCA — Principal Component Analysis}
\textit{When:} linear structure; want interpretable, fast compression.

\begin{defn}[title=Covariance Matrix]
\[ C = \frac{1}{N}\sum_{i=1}^{N}(x_i-\bar{x})(x_i-\bar{x})^T \in \R^{D\times D} \]
\end{defn}

\begin{thm}[title=PCA Solution]
$w_{(j)}$ = $j$-th eigenvector of $C$ (sorted $\lambda_1\geq\cdots\geq\lambda_d$).
\[ W = [w_{(1)}\cdots w_{(d)}] \in \R^{D\times d}, \quad W^TW = I_d \]
\textbf{Project:} $y_i = W^T(x_i-\bar{x}) \in \R^d$\\
\textbf{Reconstruct:} $\hat{x}_i = \bar{x} + Wy_i$
\end{thm}

\begin{prop}[title=Choose $d$]
Retain $\geq 90\%$ variance: $\sum_{j=1}^d\lambda_j \geq 0.9\sum_{k=1}^D\lambda_k$
\end{prop}

\begin{prop}[title=PCA minimizes reconstruction error]
\[ e_i = \|\hat{x}_i - x_i\|^2 \]
The $d$ leading eigenvectors minimize total reconstruction error.
\end{prop}

\textit{Limitation:} only captures linear structure.

\subsection{Kernel PCA}
\textit{When:} non-linear structure in data.

Map $x_i \mapsto \phi(x_i)$, apply PCA in feature space. By representer theorem, eigenvectors of covariance lie in span of $\{\phi(x_i)\}$, so:
\[ Ka = \lambda N a, \quad K_{ij}=k(x_i,x_j) \]
\[ y_i^{(j)} = \sum_{l=1}^N a_l^{(j)}\,k(x_i,x_l) \]
Center kernel: $\tilde{K} = K - \mathbf{1}_N K - K\mathbf{1}_N + \mathbf{1}_N K\mathbf{1}_N$

\textit{Limitation vs PCA:} no inverse mapping (can't reconstruct $\hat{x}$ from $y$).

\subsection{Autoencoders}
\textit{When:} non-linear structure; need encoder \& decoder; large data.

\begin{defn}[title=Autoencoder]
\textbf{Encoder:} $x \to y$ (latent representation)\\
\textbf{Decoder:} $y \to \hat{x}$\\
\textbf{Goal:} $\hat{x} \approx x$, trained by minimizing $\frac{1}{N}\sum_i\|\hat{x}_i - x_i\|^2$ with SGD.
\end{defn}

\begin{defn}[title=Variants]
\kv{Denoising}{add noise to input, train to reconstruct clean version}
\kv{Sparse}{add $\ell_1$ penalty on latent code $\Rightarrow$ sparse activations}
\end{defn}

\begin{center}
\includegraphics[width=\linewidth]{images/denoising_autoencoder.png}
\end{center}

\textit{vs PCA:} autoencoder is non-linear, more powerful; has both encoder \& decoder; non-convex optimization.\\
\textit{Applications:} compression, anomaly detection, generative models.

\cheattitle{Models}{}{}

\section{Linear Regression [REG]}
\textit{When:} output is continuous; relationship is approximately linear.\\
\textit{Output:} $\hat{y}_i \in \R$.

\begin{defn}[title=Model \& Loss (MSE)]
\[ \hat{y}_i = w^T x_i \qquad R(w) = \frac{1}{N}\sum_{i=1}^{N}(w^T x_i - y_i)^2 \]
\end{defn}

\begin{thm}[title=Closed-Form Solution]
\[ \nabla_w R = 0 \;\Rightarrow\; w^\star = (X^TX)^{-1}X^Ty \]
If $X^TX$ not invertible: $w^\star = X^\dagger y$ (pseudo-inverse).\\
\textbf{Multi-output ($C>1$):} $W^\star = X^\dagger Y$
\end{thm}

\begin{prop}[title=Prediction]
\[ \hat{y} = w^{\star T}x \quad(C{=}1) \qquad \hat{y} = W^{\star T}x \quad(C{>}1) \]
\end{prop}

\begin{warn}[title=Invertibility]
$X^TX$ invertible $\Leftrightarrow$ columns of $X$ linearly independent ($X$ full rank). Otherwise use pseudo-inverse $X^\dagger$.
\end{warn}
\textit{Pros:} exact closed-form, fast, interpretable. \textit{Cons:} only linear relationships.

\section{Logistic Regression (Binary) [CLA]}
\textit{When:} binary classification ($y_i\in\{0,1\}$); approximately linearly separable.\\
\textit{Output:} $\hat{y}_i = p(\text{class}=1\mid x_i) \in (0,1)$.

\begin{defn}[title=Model]
\[ \hat{y}_i = \sigma(w^T x_i) = \frac{1}{1+e^{-w^Tx_i}} \]
\end{defn}

\begin{defn}[title=Binary Cross-Entropy Loss]
\[ R(w) = -\sum_{i=1}^{N}\!\Big[y_i\ln\hat{y}_i + (1-y_i)\ln(1-\hat{y}_i)\Big] \]
\end{defn}

\begin{prop}[title=Gradient (no closed form $\Rightarrow$ GD)]
\[ \nabla_w R(w) = \sum_{i=1}^{N}(\hat{y}_i - y_i)\,x_i \]
\end{prop}

\begin{prop}[title=Prediction]
\[ \hat{y} = \begin{cases}1 & \sigma(w^{\star T}x)\geq 0.5\\0 & \text{otherwise}\end{cases} \]
Threshold $0.5$ can be adjusted (e.g.\ raise for high precision).
\end{prop}

\section{Multi-Class Classification [CLA]}
\textit{When:} $C>2$ classes. Labels as one-hot: $y_i\in\{0,1\}^C$.

\subsection{Linear (Least Squares)}
\[ \min_W \sum_i\|W^Tx_i - y_i\|^2 \;\Rightarrow\; W^\star = X^\dagger Y \]
\[ \hat{y}_i = W^{\star T}x_i \in \R^C, \quad k = \argmax_j\,\hat{y}_i^{(j)} \]

\subsection{Softmax (Logistic)}
\begin{defn}[title=Softmax Output]
\[ \hat{y}^{(k)}(x) = \frac{\exp(w^{(k)T}x)}{\sum_{j=1}^{C}\exp(w^{(j)T}x)} \approx p(\text{class}=k\mid x) \]
\end{defn}

\begin{defn}[title=Multi-Class Cross-Entropy]
\[ R(W) = -\sum_{i=1}^{N}\sum_{k=1}^{C} y_i^{(k)}\ln\hat{y}_i^{(k)} \]
\end{defn}

\begin{prop}[title=Gradient \& Prediction]
\[ \nabla_W R = \sum_{i=1}^{N} x_i(\hat{y}_i - y_i)^T \qquad k = \argmax_j\,\hat{y}^{(j)} \]
\end{prop}

\section{SVM --- Support Vector Machines [R/CLA]}
\textit{When:} binary classification; want maximum-margin boundary; works well in high dim.\\
\textit{Output:} $\hat{y} \in \{-1,+1\}$.

\begin{defn}[title=Key Concept: Margin]
Distance between the hyperplane $w^Tx+b=0$ and the closest points.
\[ \text{margin} = \frac{2}{\|w\|} \quad\text{(maximize} \Leftrightarrow \text{minimize }\|w\|) \]
\textbf{Support vectors:} the points defining the margin.
\end{defn}

\begin{prop}[title=Prediction]
\[ \hat{y} = \text{sign}(w^{\star T}x + b) \]
\end{prop}

\section{k-Nearest Neighbors (KNN) [R/CLA]}
\textit{When:} non-linear decision boundaries; small dataset; no training phase needed.\\
\textit{Output:} class label or continuous value.

\begin{defn}[title=Distances]
\kv{Euclidean}{$\sqrt{\sum_{d}(x_i^{(d)}-x_j^{(d)})^2}$}
\end{defn}

\begin{prop}[title=Classification]
Find $k$ nearest neighbors $\Rightarrow$ \textbf{majority vote}.
\end{prop}

\begin{prop}[title=Regression]
\[ \hat{y} = \frac{\sum_i w_i y_i}{\sum_i w_i} \quad w_i=\frac{1}{d_i} \text{ (inv. dist.)} \]
\end{prop}

\begin{defn}[title=Hyperparameter $k$]
\kv{$k=1$}{very flexible, overfitting risk}
\kv{$k$ large}{smooth, underfitting risk}
Choose $k$ by cross-validation. Use odd $k$ for 2 classes.
\end{defn}

\textit{Cons:} no training but slow prediction $O(ND)$; sensitive to scale $\Rightarrow$ \textbf{normalize first!}; sensitive to outliers.

\section{Kernel Methods [R/CLA]}
\textit{Why:} when $\phi$ is very high-dim or infinite-dim, computing $\phi(x)$ is infeasible. We only need dot products $\phi(x_i)^T\phi(x_j) = k(x_i,x_j)$.

\begin{defn}[title=Kernel Function]
\[ k(x_i,x_j) = \phi(x_i)^T\phi(x_j) \quad (\text{implicit }\phi) \]
\kv{Polynomial}{$(x_i^Tx_j + c)^d$}
\end{defn}

\begin{defn}[title=Kernel Matrix \& Vector]
\[ K \in \R^{N\times N},\; K_{ij}=k(x_i,x_j) \quad\text{(symmetric, PSD)} \]
\[ k(X,x) = (k(x_1,x),\ldots,k(x_N,x))^T \in \R^N \]
\end{defn}

\begin{thm}[title=Kernel Ridge Regression]
Reparametrize $w = \Phi^T\alpha$, solve for $\alpha \in \R^N$:
\[ \alpha^\star = (K + \lambda I_N)^{-1}y \]
\[ \hat{y} = y^T(K+\lambda I)^{-1}k(X,x) \]
\textbf{Multi-output:} $\hat{y} = Y^T(K+\lambda I)^{-1}k(X,x)$
\end{thm}

\begin{warn}[title=Kernel Selection]
Choose kernel and hyperparams ($d$, $\sigma$, $\lambda$) by \textbf{cross-validation}.
\end{warn}

\textit{Kernel vs explicit $\phi$:} kernel is $O(N^2)$ in space vs $O(NF)$

\section{Neural Networks / MLP [R/CLA]}
\textit{When:} large datasets; complex non-linear structure; learns features automatically.\\
\textit{Output:} any type depending on final activation.

\subsection{Architecture (Forward Pass)}
\begin{defn}[title=Layers]
\[ z_{(1)} = f(W_{(1)}^T x) \quad z_{(l)} = f(W_{(l)}^T z_{(l-1)}) \]
\[ \hat{y} = W_{(L)}^T z_{(L-1)} \quad\text{(or softmax for classification)} \]
\end{defn}

\begin{defn}[title=Activation Functions]
\kv{ReLU}{$\max(0,z)$} — hidden layers (most common)
\kv{Sigmoid}{$\frac{1}{1+e^{-z}}$} — binary output
\kv{Softmax}{$\frac{e^{z^{(k)}}}{\sum_j e^{z^{(j)}}}$} — multi-class output
\end{defn}

\begin{thm}[title=Universal Approximation]
Any continuous function can be approximated by a linear combination of translated/scaled ReLU (or sigmoid) functions.
\end{thm}

\subsection{Training: Backpropagation}
\begin{defn}[title=Loss Functions]
\kv{Regression}{$R(W)=\frac{1}{N}\sum_i\|\hat{y}_i-y_i\|^2$}
\kv{Classification}{$R(W)=-\frac{1}{N}\sum_i\sum_k y_i^{(k)}\ln\hat{y}_i^{(k)}$}
\end{defn}

\begin{prop}[title=Backprop Algorithm]
\textbf{1.} Forward pass: compute all $z_{(l)}$ and $\hat{y}$.\\
\textbf{2.} Compute output sensitivity $\delta^{(k)}_{(L)} = \frac{\partial\ell_i}{\partial a^{(k)}_{(L)}}$.\\
\textbf{3.} Propagate backward: $\delta_{(l)} \to \delta_{(l-1)}$ via chain rule.\\
\textbf{4.} Gradient: $\frac{\partial\ell_i}{\partial W^{(k,j)}_{(l)}} = \delta^{(k)}_{(l)}\,z^{(j)}_{(l-1)}$.
\end{prop}

\subsection{SGD \& Variants}
\begin{defn}[title=Mini-Batch SGD]
\[ W_k \leftarrow W_{k-1} - \eta\sum_{b=1}^{B}\nabla\ell_{(b,k)}(W_{k-1}) \]
$B$ = batch size. Full GD: $B=N$; SGD: $B=1$; mini-batch: $1<B<N$.
\end{defn}

\begin{defn}[title=SGD with Momentum]
\[ \nu \leftarrow \gamma\nu + \eta\nabla R(W) \qquad W \leftarrow W - \nu \]
$\gamma>0$ adds inertia; accelerates if gradient is consistent.
\end{defn}

\textit{Hyperparams:} learning rate $\eta$, batch size $B$, depth $L$, width $M$, activation.\\
\textit{Overfitting:} use dropout, weight decay (L2), data augmentation, early stopping.

\section{Convolutional Neural Networks [REG/CLA]}
\textit{When:} image/spatial data; translation-invariant features; large datasets.\\
\textit{Output:} any type depending on final layer (same as MLP).

\begin{defn}[title=Architecture]
\textbf{Conv} $\to$ \textbf{ReLU} $\to$ \textbf{Pool} $\to$ \ldots $\to$ \textbf{Flatten} $\to$ \textbf{FC layers} $\to$ output\\
\kv{Conv layer}{learns spatial filters; shares weights $\Rightarrow$ fewer params}
\kv{Pooling}{reduces spatial size; max pooling keeps strongest activation}
\kv{FC layer}{standard MLP at the end for final prediction}
\end{defn}

\begin{defn}[title=Hyperparameters]
\kv{Filter size}{e.g.\ $3\times3$, $5\times5$}
\kv{Stride}{step size of the sliding filter}
\kv{Padding}{preserve spatial dims (same) or reduce (valid)}
\kv{Depth}{number of filters per layer}
\end{defn}

\textit{Key advantage over MLP:} weight sharing + local connectivity $\Rightarrow$ translation invariance, far fewer parameters for image inputs.\\
\textit{Training:} backprop + SGD, same as MLP.

\section{K-Means Clustering [CLU]}
\textit{When:} unsupervised; want to partition $N$ unlabeled points into $K$ groups.

\begin{prop}[title=Algorithm]
\textbf{1.} Initialize $K$ centroids $\mu_1,\ldots,\mu_K$ randomly.\\
\textbf{2a. Assign:} $c_i = \argmin_k\|x_i - \mu_k\|^2$ for all $i$.\\
\textbf{2b. Update:} $\mu_k = \frac{1}{|C_k|}\sum_{i:c_i=k} x_i$.\\
\textbf{3.} Repeat until assignments don't change.
\end{prop}

\begin{warn}[title=Key Caveats]
\kv{Convergence}{guaranteed, but may reach local minimum (not global)}
\kv{Initialization}{run multiple times with different random inits}
\kv{Scale}{always normalize features --- K-Means uses Euclidean distance!}
\kv{Choose $K$}{elbow method on within-cluster variance}
\end{warn}

\textit{Complexity:} $O(NKD)$ per iteration.\\
\textit{Limitation:} assumes spherical clusters; sensitive to outliers.

\section{Evaluation --- Regression}

\begin{defn}[title=Regression Metrics]
\kv{MSE}{$\frac{1}{N}\sum(y_i-\hat{y}_i)^2$} — penalizes large errors
\kv{RMSE}{$\sqrt{\text{MSE}}$} — same unit as $y$; more interpretable
\kv{MAE}{$\frac{1}{N}\sum|y_i-\hat{y}_i|$} — robust to outliers
\kv{MAPE}{$\frac{1}{N}\sum\left|\frac{y_i-\hat{y}_i}{y_i}\right|$} — relative; undef.\ if $y_i{=}0$
\end{defn}

\begin{defn}[title=Interpretation]
\kv{MSE$\approx 0$}{near-perfect predictions}
\kv{RMSE}{error in the unit of $y$, e.g.\ RMSE=5€ means avg error of 5€}
\kv{MAE vs MSE}{MAE if outliers shouldn't dominate; MSE if large errors are critical}
\end{defn}

\section{Evaluation --- Classification}

\begin{defn}[title=Confusion Matrix (binary)]
\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\linewidth}{@{}lXX@{}}
\toprule
 & \textbf{Pred Pos} & \textbf{Pred Neg}\\
\midrule
\textbf{True Pos} & TP & FN (type II)\\
\textbf{True Neg} & FP (type I) & TN\\
\bottomrule
\end{tabularx}
\[ P = TP+FN \quad N = FP+TN \]
\end{defn}

\begin{defn}[title=Metrics]
\kv{Accuracy}{$\frac{TP+TN}{TP+TN+FP+FN}$}
\kv{Precision}{$\frac{TP}{TP+FP}$}
\kv{Recall}{$\frac{TP}{TP+FN}$}
\kv{FPR}{$\frac{FP}{FP+TN}$}
\kv{F1}{$2\cdot\frac{\text{Prec}\cdot\text{Rec}}{\text{Prec}+\text{Rec}}$}
\end{defn}

\begin{defn}[title=Interpretation --- Goal $\to$ Metric]
\kv{Reduce FP}{$\uparrow$ Precision (spam filter, recommendations)}
\kv{Reduce FN}{$\uparrow$ Recall (cancer detection, fraud)}
\kv{Balance FP \& FN}{$\uparrow$ F1 (imbalanced classes)}
\kv{Reduce FPR}{$\downarrow$ FP/(FP+TN): fewer false alarms on negatives}
\kv{Imbalanced data}{F1 or weighted avg; \textit{Accuracy misleading!}}
\end{defn}

\textbf{Multi-class:} $C\times C$ confusion matrix. Compute precision/recall per class (one-vs-all) then average.
\begin{itemize}
  \item \kv{Macro avg}{equal weight per class}
  \item \kv{Weighted avg}{weight by class size; use if imbalanced}
\end{itemize}

\section{Evaluation --- Clustering}

\textbf{Without labels} (unsupervised):

\begin{defn}[title=Inertia (Within-cluster variance)]
\[ W = \sum_{k=1}^{K}\sum_{i:c_i=k}\|x_i - \mu_k\|^2 \]
Lower $W$ = tighter clusters.
\end{defn}

\begin{prop}[title=Choose $K$ --- Elbow Method]
Plot $W$ vs $K$. Choose $K$ at the "elbow" where gain diminishes.
\end{prop}

\textbf{With labels} (supervised): training is unsupervised but if ground truth labels are available, Accuracy, Precision, Recall, F1 apply identically to classification.

\section{Overfitting, Underfitting \& CV}

\begin{defn}[title=Bias-Variance]
\kv{Underfitting}{high error on train \& test (model too simple)}
\kv{Overfitting}{low train error, high test error (model too complex)}
\end{defn}

\begin{defn}[title=Cross-Validation Strategies]
\kv{Val.\ set}{80/20 split; fast but estimate depends on random split}
\kv{K-fold}{split in $K$ parts, rotate; $K{=}5$ or $10$ is standard}
\kv{LOOCV}{$K=N$; exact but very expensive}
\end{defn}

\textit{Rule:} never use test set to choose hyperparameters --- use validation/CV only.
